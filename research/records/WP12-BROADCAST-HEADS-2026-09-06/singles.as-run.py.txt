"""Single-head ablation of every in-band attention head (R41e attention members, derived from the model's layer kinds):
each of the 80 heads zeroed alone, the lens read at all ten band layers on the same documents and positions as ablation.py,
against the unablated readout. Per head: top-25 overlap with the unablated readout at every readout layer, its mean over the
layers at or above the head (downstream), the overlap at layer 28 (downstream of every head, the one readout common to all),
and the next-token change rate. Purpose: does stage-1 preservation order heads by causal effect? Selected against unselected
at each layer, and the rank correlation of preservation with single-head damage over all 80 heads. Writes out/singles.json."""
import json, os, sys, time
from pathlib import Path
import numpy as np
HERE = Path(__file__).resolve().parent; OUT = HERE / "out"; LENS = Path(sys.argv[1])
DOCS, POS, DOC_TOKENS = 8, 64, 512
BAND_PAIRS = ((12, 13), (16, 17), (19, 20), (23, 24), (27, 28))   # R41e, asserted as the recorded ruling (R38(f))
BAND = tuple(sorted(L for pr in BAND_PAIRS for L in pr))
import mlx.core as mx
mx.set_cache_limit(2 * 1024 ** 3)
from mlx_lm.models import qwen3_next as q3n
from local_llm_lab.models import load_model_spec
from local_llm_lab.pipeline.evaluate import load_policy
from local_llm_lab.pipeline.jlens import encode
T0 = time.time(); spec = load_model_spec("qwen35-4b"); model, tokenizer, view, resolved = load_policy(spec, None); model.eval()
att_blocks = [b for b, l in enumerate(model.language_model.model.layers) if not l.is_linear]
BAND_ATT = tuple(L for L in BAND if (L - 1) in att_blocks); assert len(BAND_ATT) == 5, BAND_ATT
lens = np.load(LENS); J = {L: mx.array(lens[f"J{L - 1}"].astype(np.float32)) for L in BAND}
ABLATE = {}; _orig = q3n.scaled_dot_product_attention; _ctr = {"i": 0}
def hooked(queries, keys, values, cache, scale, mask, *a, **kw):
    out = _orig(queries, keys, values, cache, scale, mask, *a, **kw)
    b = att_blocks[_ctr["i"] % len(att_blocks)]; _ctr["i"] += 1; hs = ABLATE.get(b)
    if hs:
        m = np.ones((1, out.shape[1], 1, 1), dtype=np.float32); m[0, list(hs)] = 0.0; out = out * mx.array(m).astype(out.dtype)
    return out
q3n.scaled_dot_product_attention = hooked
corpus = json.load(open(HERE.parent / "retrieval" / "corpus.json")); docs = [encode(tokenizer, c["text"])[:DOC_TOKENS] for c in corpus[:DOCS]]
rng = np.random.default_rng(0); positions = [sorted(rng.choice(np.arange(16, len(d)), POS, replace=False).tolist()) for d in docs]
def run(ablate):
    ABLATE.clear(); ABLATE.update(ablate); tops, top1 = {L: [] for L in BAND}, []
    for d, pos in zip(docs, positions):
        _ctr["i"] = 0; res = view.residuals(d, list(BAND)); _ctr["i"] = 0
        logits = model(mx.array(d)[None])[0]; top1.append(np.array(mx.argmax(logits[pos], axis=-1)).tolist())
        for L in BAND:
            h = res[L][0][pos] if res[L].ndim == 3 else res[L][pos]
            read = view.unembed(view.final_norm(h.astype(mx.float32) @ J[L].T)); tops[L].append(np.array(mx.argsort(read, axis=-1)[:, -25:]).tolist())
    mx.eval(); return tops, top1
def overlap(base, test): return {L: float(np.mean([len(set(a) & set(b)) / 25 for da, db in zip(base[L], test[L]) for a, b in zip(da, db)])) for L in BAND}
def change(a, b): return float(np.mean([x != y for da, db in zip(a, b) for x, y in zip(da, db)]))
base_tops, base_top1 = run({}); rep_tops, rep_top1 = run({}); identical = all(overlap(base_tops, rep_tops)[L] == 1.0 for L in BAND) and change(base_top1, rep_top1) == 0.0
print(json.dumps({"event": "reproducibility", "bit_identical": identical, "elapsed_s": round(time.time() - T0, 1)}), flush=True)
sel = json.load(open(OUT / "selection_orth.json")); pres = {(r["block"], r["head"]): r["J_mrr"] for r in json.load(open(OUT / "broadcast_heads_orth.json"))["rows"] if r["channel"] == "attention"}
selected = {(r["block"], r["head"]) for r in sel["selected_attention"]}
rows = []
for L in BAND_ATT:
    b = L - 1
    for h in range(16):
        tops, t1 = run({b: {h}}); ov = overlap(base_tops, tops); down = [ov[x] for x in BAND if x >= L]; up = {x: ov[x] for x in BAND if x < L}
        rows.append({"block": b, "head": h, "layer_written": L, "selected": (b, h) in selected, "preservation": pres[(b, h)], "overlap_by_layer": ov, "downstream_mean": float(np.mean(down)), "overlap_at_28": ov[28], "upstream_exact": all(v == 1.0 for v in up.values()), "change_rate": change(base_top1, t1)})
        json.dump({"rows": rows, "partial": True}, open(OUT / "singles.json", "w"), indent=1)
    done = [r for r in rows if r["layer_written"] == L]; s = [r for r in done if r["selected"]]; u = [r for r in done if not r["selected"]]
    print(json.dumps({"event": "layer", "layer": L, "selected_n": len(s), "selected_median_overlap_at_28": round(float(np.median([r["overlap_at_28"] for r in s])), 3) if s else None, "unselected_median_overlap_at_28": round(float(np.median([r["overlap_at_28"] for r in u])), 3), "upstream_exact_all": all(r["upstream_exact"] for r in done), "elapsed_s": round(time.time() - T0, 1)}), flush=True)
from scipy.stats import spearmanr
rho_all = spearmanr([r["preservation"] for r in rows], [1 - r["overlap_at_28"] for r in rows]); rho_sel = spearmanr([r["preservation"] for r in rows if r["selected"]], [1 - r["overlap_at_28"] for r in rows if r["selected"]])
summary = {"spearman_preservation_vs_damage_at_28_all_80": [float(rho_all.correlation), float(rho_all.pvalue)], "spearman_within_selected": [float(rho_sel.correlation), float(rho_sel.pvalue)], "reproducibility_bit_identical": identical, "readout_layers": BAND, "ablated_layers": BAND_ATT, "elapsed_s": round(time.time() - T0, 1)}
json.dump({"rows": rows, "summary": summary}, open(OUT / "singles.json", "w"), indent=1); print(json.dumps({"event": "done", **summary}), flush=True)
