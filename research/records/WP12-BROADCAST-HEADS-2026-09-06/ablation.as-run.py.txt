"""WP12 stage 2: graded ablation of the lens-relay heads (the Head's design, 03:50).

Ablate the top k attention heads by preservation margin (k in K_LIST, and the full selected set),
zeroing each head's attention output at every position before the output gate and o_proj, against
layer-matched random sets of the same size (NSEED seeds), on DOCS documents of POS sampled
positions each. Measures, per k and per set: recall@25 of the hosted-lens readout at the band
layers (the fraction of the unablated run's top-25 tokens at a position that remain in the top 25),
and the next-token change rate (the fraction of positions whose top-1 prediction changes). The
threshold, stated in advance: a recall drop counts only if the change rate stays within twice the
layer-matched random control's at the same k."""

def run(ablate):
    """residuals at the band layers and logits at sampled positions; returns top-25 lens tokens per (layer, doc, pos) and top-1 next tokens."""
    ABLATE.clear(); ABLATE.update(ablate); tops, top1 = {L: [] for L in BAND}, []

            read = view.unembed(view.final_norm(h.astype(mx.float32) @ J[L].T))
            tops[L].append(np.array(mx.argsort(read, axis=-1)[:, -25:]).tolist())
    mx.eval(); return tops, top1
def recall(base, test):
    return {L: float(np.mean([len(set(a) & set(b)) / 25 for da, db in zip(base[L], test[L]) for a, b in zip(da, db)])) for L in BAND}
UPSTREAM_FLOOR = {"recall": 1.0, "change": 0.0}   # set from the reproducibility pass below: the gate is on recall@25 and the top-1 change rate, exact equality if the unablated readout reproduces bit-identically
def stratify(heads, rec):
    """recall at readout layers with at least one ablated head writing at or below them (the measurement), and at
    layers with none (the control, which must be unchanged); the number of upstream ablated heads per layer beside it."""
    written = [b + 1 for b, h in heads]
    up = {L: sum(1 for w in written if w <= L) for L in BAND}
    above = {L: rec[L] for L in BAND if up[L] > 0}; below = {L: rec[L] for L in BAND if up[L] == 0}
    if any(v < UPSTREAM_FLOOR["recall"] for v in below.values()):
        raise SystemExit(f"upstream control failed: recall@25 at readout layers with no ablated head at or below them fell below the reproducibility floor {UPSTREAM_FLOOR['recall']}: {below}")
    return {"upstream_heads_per_layer": up, "recall_downstream": above, "recall_downstream_mean": float(np.mean(list(above.values()))) if above else None, "recall_upstream_control": below}
def change(base, test): return float(np.mean([a != b for da, db in zip(base, test) for a, b in zip(da, db)]))

def by_block(heads):
    d = {}
    for b, h in heads: d.setdefault(b, set()).add(h)
    return d
CONTROL = os.environ.get("CONTROL", "exclude")   # "exclude": unselected heads at the layer where possible, forced overlap only where the pool is short; "all": the paper's draw from all sixteen
def layer_matched_random(heads, seed):
    """Equal-sized random sets at the same layers. Under CONTROL=exclude the draw avoids the selection where
    the layer's unselected pool is large enough and takes the shortfall from the selection (forced overlap,
    counted); under CONTROL=all it draws from all sixteen heads. Both biases are stated in the record: the
    exclude draw is the weakest heads at a saturated layer, the all draw is more than half treatment there."""
    r = np.random.default_rng(seed); out = []; sel = set(map(tuple, heads))
    for b, hs in by_block(heads).items():
        if CONTROL == "all":
            out += [(b, int(h)) for h in r.choice(16, len(hs), replace=False)]; continue
        pool = [h for h in range(16) if (b, h) not in sel]; n = len(hs)
        take = [int(h) for h in r.choice(pool, min(n, len(pool)), replace=False)]
        if n > len(pool): take += [int(h) for h in r.choice(sorted(hs), n - len(pool), replace=False)]
        out += [(b, h) for h in take]
    return out
