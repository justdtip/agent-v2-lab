"""WP12 stage 1: the paper's broadcast-head metrics (§4.3.2) on Qwen3.5-4B, from weights and the hosted lens.

For every attention head (8 blocks x 16 query heads; GQA over 4 kv heads) the weight-only OV map
M_h v = W_O[:, h] (W_V[kv(h)] v), a fixed linear map from residual to residual; this omits the
attention pattern (by construction) and the block's sigmoid output gate. For every recurrent value
path (24 blocks x 32 value heads) the weight-only composition M_h v = out_proj[:, h] (in_proj_v[h] v),
which OMITS the causal conv1d, its silu, the gated-delta state and the gated RMSNorm before out_proj,
all input-dependent: the two channels' columns are different objects and are reported apart.

Populations at the block's input layer L = b (file key J{b-1}; layer L = residual after block L-1;
the lens maps h -> J h so the direction for token w is J^T u_w with u_w the tied embedding row):
J (2,000 sampled vocabulary tokens), J-rotated (the same vectors under one fixed random orthogonal
rotation), MLP rows (2,000 down_proj output directions of the block before b), and isotropic random
directions for normalisation. Metrics per head and population, as in the paper: gain = mean ||M v||
over the population divided by the median ||M r|| over random unit r; label preservation = mean
reciprocal rank of cos(M v_i, v_i) among {cos(M v_i, v_j)}_j, with the same on random directions.

Selection rule, fixed in advance (the Head, WP12 amendment 1): no percentile. A head "broadcasts J"
when its gain on J exceeds its gain on BOTH controls and its label preservation on J exceeds BOTH
controls'; the count is reported as n of N per channel and the whole list is written ordered by the
worse of the two J-minus-best-control margins (in units of the control's value)."""

def dense(linear):
    """Weight as a float32 (out, in) matrix, dequantised when the layer is quantised."""
    w = linear.weight
    if hasattr(linear, "scales"):
        w = mx.dequantize(w, linear.scales, linear.biases, linear.group_size, linear.bits)
    return w.astype(mx.float32)

tokens = mx.array(rng.choice(V, NPOP, replace=False))
Q_rot, _ = np.linalg.qr(rng.standard_normal((H, H)).astype(np.float32)); Q_rot = mx.array(Q_rot)
def unit(x): return x / (mx.linalg.norm(x, axis=1, keepdims=True) + 1e-12)
def populations(b):
    """J, J-rotated, MLP rows and random unit directions at the input layer of block b."""
    Jl = mx.array(lens[f"J{b - 1}"].astype(np.float32)) if b >= 1 else None
    U = embed[tokens]                                                    # (NPOP, H)
    J = unit(U @ Jl) if Jl is not None else unit(U)                      # rows u_w^T J = (J^T u_w)^T
    Jrot = unit(J @ Q_rot.T)
    prev = layers[b - 1] if b >= 1 else layers[0]
    mlp = getattr(prev, "mlp", None)
    down = dense(mlp.down_proj) if mlp is not None and hasattr(mlp, "down_proj") else None    # (H, inter)
    cols = mx.array(rng.choice(down.shape[1], NPOP, replace=False)) if down is not None else None
    M = unit(down[:, cols].T) if down is not None else unit(mx.random.normal((NPOP, H)))
    R = unit(mx.random.normal((NPOP, H)))
    return {"J": J, "J_rot": Jrot, "MLP_rows": M, "random": R}
def metrics(apply, pops):
    """apply: (NPOP, H) -> (NPOP, H) images under the head's map."""
    out = {}
    imgR = apply(pops["random"]); med_r = float(mx.median(mx.linalg.norm(imgR, axis=1)).item()) + 1e-12
    for name, P in pops.items():
        img = apply(P); gain = float(mx.mean(mx.linalg.norm(img, axis=1)).item()) / med_r
        C = unit(img) @ P.T                                              # cos(M v_i, v_j)
        diag = mx.diagonal(C)[:, None]
        rank = mx.sum(C > diag, axis=1) + 1                              # 1 = own label best preserved
        out[name] = {"gain": gain, "mrr": float(mx.mean(1.0 / rank.astype(mx.float32)).item())}
    return out
