caps = []
_orig = q35.gated_delta_update
def hooked(q, k, v, a, b, A_log, dt_bias, state=None, mask=None, use_kernel=True):
    out, st = _orig(q, k, v, a, b, A_log, dt_bias, state, mask, use_kernel); mx.eval(out)
    caps.append({"q": q, "k": k, "a": a, "b": b, "A_log": A_log, "dt_bias": dt_bias}); return out, st
q35.gated_delta_update = hooked
def alpha_batch(Q, kk, g, beta, t):
    W = Q.copy(); out = np.zeros((t + 1, Q.shape[0]))
    for r in range(t, -1, -1):
        kr = kk[r]; d = W @ kr; out[r] = beta[r] * d; W = g[r] * (W - beta[r] * d[:, None] * kr[None, :])
    return out
T0 = time.time(); spec = load_model_spec("qwen35-4b"); model, tokenizer, view, resolved = load_policy(spec, None); model.eval()
corpus = json.load(open(HERE / "corpus.json")); rng = np.random.default_rng(0)
rows = []
for ctx in corpus:
    for qi, qry in enumerate(ctx["queries"]):
        if qry.get("control"): continue
        ids = encode(tokenizer, ctx["text"] + qry["query"]); t = len(ids) - 1; n_elig = t - GAP + 1

            beta = np.array(mx.sigmoid(cap["b"]).astype(mx.float32))[0]; g = np.array(gd.compute_g(cap["A_log"], cap["a"], cap["dt_bias"]).astype(mx.float32))[0]

                rows.append({"kind": ctx["kind"], "context": ctx["index"], "query": qi, "block": b, "key_head": kh, "head": h, "T": t + 1, "n_elig": int(n_elig),

# overlap across DIFFERENT contexts, per head, in absolute positions (habit at fixed offsets from the start) and in bins of context fraction (habit at relative positions)
def overlap(a, b): return len(set(a) & set(b))
summary = {}
for (b, kh) in CANDS:
    for h in (kh * 2, kh * 2 + 1):
        rs = [r for r in rows if r["block"] == b and r["head"] == h]
        real_abs, null_abs, real_frac, null_frac, same_ctx = [], [], [], [], []
        for i in range(len(rs)):
            for j in range(i + 1, len(rs)):
                x, y = rs[i], rs[j]
                if (x["kind"], x["context"]) == (y["kind"], y["context"]):
                    same_ctx.append(overlap(x["top_abs"], y["top_abs"])); continue
                real_abs.append(overlap(x["top_abs"], y["top_abs"]))
                null_abs.append(np.mean([overlap(x["null_tops_abs"][d], y["null_tops_abs"][d]) for d in range(NDRAW)]))
                fx = [int(f * 20) for f in x["top_frac"]]; fy = [int(f * 20) for f in y["top_frac"]]
                real_frac.append(len(set(fx) & set(fy)))

                null_frac.append(np.mean([len(set(nx[d]) & set(ny[d])) for d in range(NDRAW)]))
