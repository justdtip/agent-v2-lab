def alpha_all_heads(q, k, g, beta, t, Hk):
    T, Hv = g.shape; rep = Hv // Hk; kk = np.repeat(k, rep, axis=1); w = np.repeat(q, rep, axis=1)[t].copy(); out = np.zeros((t + 1, Hv))
    for r in range(t, -1, -1):
        kr = kk[r]; d = np.einsum("hd,hd->h", kr, w); out[r] = beta[r] * d; w = g[r][:, None] * (w - beta[r][:, None] * kr * d[:, None])
    return out
def alpha_batch(Q, kk, g, beta, t):
    W = Q.copy(); out = np.zeros((t + 1, Q.shape[0]))
    for r in range(t, -1, -1):
        kr = kk[r]; d = W @ kr; out[r] = beta[r] * d; W = g[r] * (W - beta[r] * d[:, None] * kr[None, :])
    return out
def softmax_rows(s): m = s.max(axis=-1, keepdims=True); e = np.exp(s - m); return e / e.sum(axis=-1, keepdims=True)

    return out
